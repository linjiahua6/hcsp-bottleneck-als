    \begin{enumerate}[label=\textcolor{PosterBlue}{\textbf{\arabic*.}},leftmargin=1.25em,itemsep=0.30em,parsep=0pt,topsep=0pt,partopsep=0pt]
      \item \textbf{Task:} Heterogeneous Computing Scheduling Problem (HCSP): schedule $n$ independent tasks on $m$ heterogeneous machines to minimize the longest machine load (NP-hard).
      \item \textbf{Gap:} greedy heuristics are fast but weak; metaheuristics are strong but costly and start-sensitive.
      \item \textbf{Contribution:} BLS---a lightweight, deterministic post-processor for any initial schedule.
      \item \textbf{Insight:} gains concentrate on the bottleneck machine; the BLS advantage is larger at higher observed $\rho$.
    \end{enumerate}}

  % ---------- 02 / Method ----------
  \PosterSection{403}{321}{02}{Method / Proposed Approach}
  \PosterText{30}{470}{300}{250}{40}{50}{%
    \begin{enumerate}[label=\textcolor{PosterBlue}{\textbf{\arabic*.}},leftmargin=1.25em,itemsep=0.42em,parsep=0pt,topsep=0pt,partopsep=0pt]
      \item \textbf{Round loop:} locate bottleneck $b=\arg\max_j L_j$; take its $K$ longest tasks; per task $R$ migration targets; per target $S$ swap partners.
      \item \textbf{Acceptance:} strictly improving moves only ($C' < C - \mathrm{tol}$); output never worse than input.
      \item \textbf{Efficiency:} $n(m-1) \to \le KR(1+S)$ evaluations per round; default $(4,2,4)$; $O(n+m)$ memory; deterministic.
      \item \textbf{DR extension:} perturbation escape for trap instances; activated only when no improving move exists.
    \end{enumerate}}
